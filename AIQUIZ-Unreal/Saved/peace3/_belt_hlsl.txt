// Faithful inline port of AIQUIZ-Godot/shaders/conveyor_belt_floor.gdshader
// (flat-belt path). Inputs: WP (WorldPosition), N (VertexNormalWS), Time.
// Output: float4(albedo.rgb, rim_band*top_mask) -> drives metallic/roughness.
// Unreal world is x100 vs Godot units and axis-swapped: Godot z = belt length =
// Unreal X; Godot x = belt width = Unreal Y; up = Unreal Z.
float gz = WP.x / 100.0;
float gx = WP.y / 100.0;
float top_mask = smoothstep(0.55, 0.85, abs(N.z));
float ax = abs(gx);

float scroll_sign = 1.0;
float scroll_z = Time * 5.0;            // ~Godot _active_wall_speed
float belt_scale = 0.22;
float stripe_scale = 12.0;
float stripe_softness = 0.08;
float groove_strength = 0.12;

float belt_v = (gz + scroll_z * scroll_sign) * belt_scale;
float stripe_wave = sin(belt_v * stripe_scale);
float groove_wave = sin(belt_v * stripe_scale * 2.05);
float stripe = smoothstep(-stripe_softness, stripe_softness, stripe_wave);
float groove = 0.5 + 0.5 * groove_wave;

float3 base_color = float3(0.40, 0.41, 0.42);
float3 stripe_color = float3(0.34, 0.345, 0.35);
float3 side_color = float3(0.33, 0.34, 0.35);
float3 rim_color = float3(0.26, 0.265, 0.27);

float3 belt_col = lerp(base_color, stripe_color, stripe);
belt_col *= lerp(1.0 - groove_strength, 1.0, groove);

float fp = frac(belt_v * 7.5);
float roller_line = exp(-fp * fp * 180.0);
belt_col *= lerp(1.0 - 0.22, 1.0, 1.0 - roller_line);

float rim_inner_x = 11.15;
float rim_outer_x = 12.0;
float rim_softness = 0.045;
float rim_lip_width = 0.03;

float rim_band = smoothstep(rim_inner_x - rim_softness, rim_inner_x + rim_softness * 0.35, ax);
rim_band *= 1.0 - smoothstep(rim_outer_x - rim_softness * 0.4, rim_outer_x + 0.01, ax);
float3 with_rim = lerp(belt_col, rim_color, rim_band * top_mask);

float lipx = (ax - rim_inner_x) / rim_lip_width;
float lip = exp(-lipx * lipx) * top_mask;
with_rim = lerp(with_rim, rim_color * 0.88, saturate(lip * 0.9));

float outer_edge = smoothstep(rim_outer_x - 0.18, rim_outer_x - 0.02, ax);
outer_edge *= 1.0 - smoothstep(rim_outer_x - 0.02, rim_outer_x + 0.02, ax);
with_rim = lerp(with_rim, float3(0.48, 0.49, 0.50), outer_edge * top_mask * 0.35);

float3 final_col = lerp(side_color, with_rim, top_mask);
float rim_meta = rim_band * top_mask;
return float4(final_col, rim_meta);
