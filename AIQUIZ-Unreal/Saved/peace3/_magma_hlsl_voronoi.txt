// Realistic basaltic lava (pahoehoe/aa): dark solidified crust dominates the
// surface, with concentrated glowing veins/channels along the plate boundaries.
// Polarity is INVERTED vs. the old Godot port: heat is ~0 on the crust and only
// rises inside the cracks, so the deep dark color fills most of the screen.
//
// This node also outputs a tangent-space NORMAL derived analytically from a
// height field (raised plates + micro bumps, sunken crack channels) so the dark
// crust catches directional/sky light and reads as a solid relief surface.
//
// Custom node constraints: no helper funcs / no forward refs -> everything is
// inlined. The height field is evaluated at the centre UV plus two neighbours
// (an [unroll] loop over 3 samples) for a central-difference normal.
//
// Inputs: UV(float2), Time(float), VScale, VeinWidth, FbmScale, CrustEmber,
//         NormalStrength (scalars).
// Additional output: OutNormal(float3, tangent space).
// Return: float4(veinColor.rgb, heat).

float Tw = fmod(Time, 120.0);          // keep coords small -> stable analytic noise

float vScale     = VScale;
float veinWidth  = VeinWidth;
float fbmScale   = FbmScale;
float crustEmber = CrustEmber;
float flowSpeed  = 0.12;               // slow lava drift
float veinDepth  = 0.32;               // how deep the crack channels sink

float2 flow = float2(Tw * flowSpeed, -Tw * flowSpeed * 0.7);

// 3 sample points: centre + epsilon in U and V (for the central-difference normal)
float e = 1.0 / 2048.0;
float2 SUV[3] = { UV, UV + float2(e, 0.0), UV + float2(0.0, e) };
float Hs[3];

float heatC = 0.0;                      // captured at the centre sample (s==0)
float vtC   = 0.0;

[unroll] for (int s = 0; s < 3; s++) {
    float2 uv = SUV[s];

    // --- primary edge Voronoi (plate boundaries) ---
    float2 uvc = uv * vScale + flow;
    float2 g  = floor(uvc);
    float2 fpp = frac(uvc);
    float F1 = 8.0, F2 = 8.0;
    [unroll] for (int j = -1; j <= 1; j++) {
    [unroll] for (int i = -1; i <= 1; i++) {
        float2 b = float2(i, j);
        float2 q = g + b;
        float2 pnt = frac(sin(float2(dot(q, float2(127.1, 311.7)), dot(q, float2(269.5, 183.3)))) * 43758.5453);
        pnt = 0.5 + 0.5 * sin(Tw * 0.5 + 6.2831 * pnt);
        float2 r = b - fpp + pnt;
        float d = dot(r, r);
        if (d < F1) { F2 = F1; F1 = d; } else if (d < F2) { F2 = d; }
    } }
    float edge = sqrt(F2) - sqrt(F1);

    // --- secondary finer Voronoi (sub-cracks) ---
    float2 uvc2 = uv * (vScale * 2.4) + flow * 1.7;
    float2 g2  = floor(uvc2);
    float2 fp2 = frac(uvc2);
    float G1 = 8.0, G2 = 8.0;
    [unroll] for (int j = -1; j <= 1; j++) {
    [unroll] for (int i = -1; i <= 1; i++) {
        float2 b = float2(i, j);
        float2 q = g2 + b;
        float2 pnt = frac(sin(float2(dot(q, float2(127.1, 311.7)), dot(q, float2(269.5, 183.3)))) * 43758.5453);
        pnt = 0.5 + 0.5 * sin(Tw * 0.4 + 6.2831 * pnt);
        float2 r = b - fp2 + pnt;
        float d = dot(r, r);
        if (d < G1) { G2 = G1; G1 = d; } else if (d < G2) { G2 = d; }
    } }
    float edge2 = sqrt(G2) - sqrt(G1);

    float edgeC = min(edge, edge2 * 1.3 + 0.02);

    float aa = fwidth(edgeC) * 1.5;
    float vein     = 1.0 - smoothstep(0.0, max(veinWidth, aa), edgeC);
    float veinCore = 1.0 - smoothstep(0.0, max(veinWidth * 0.4, aa), edgeC);

    // --- fbm value-noise (crust micro texture) ---
    float amp = 0.5, fr = 1.0, fbm = 0.0;
    [unroll] for (int o = 0; o < 5; o++) {
        float2 p = uv * fbmScale * fr + float2(Tw * 0.04, -Tw * 0.03);
        float2 ip = floor(p); float2 f = frac(p); f = f * f * (3.0 - 2.0 * f);
        float a = frac(sin(dot(ip,                float2(127.1, 311.7))) * 43758.5453);
        float b = frac(sin(dot(ip + float2(1, 0), float2(127.1, 311.7))) * 43758.5453);
        float c = frac(sin(dot(ip + float2(0, 1), float2(127.1, 311.7))) * 43758.5453);
        float d = frac(sin(dot(ip + float2(1, 1), float2(127.1, 311.7))) * 43758.5453);
        fbm += amp * lerp(lerp(a, b, f.x), lerp(c, d, f.x), f.y);
        fr *= 2.0; amp *= 0.5;
    }

    // --- height: plates raised + micro bumps, crack channels sunken ---
    float height = (1.0 - vein) * (0.5 + 0.5 * fbm) + (fbm - 0.5) * 0.25 - vein * veinDepth;
    Hs[s] = height;

    if (s == 0) {
        // large-scale thermal regions: some patches freshly molten (bright,
        // toward white-hot), others cooling (dim, deep red). Breaks the uniform
        // "neon web" look into an organic temperature field.
        float2 rp = uv * 2.5 + float2(Tw * 0.02, Tw * 0.015);
        float2 rip = floor(rp); float2 rf = frac(rp); rf = rf * rf * (3.0 - 2.0 * rf);
        float ra = frac(sin(dot(rip,                float2(127.1, 311.7))) * 43758.5453);
        float rb = frac(sin(dot(rip + float2(1, 0), float2(127.1, 311.7))) * 43758.5453);
        float rc = frac(sin(dot(rip + float2(0, 1), float2(127.1, 311.7))) * 43758.5453);
        float rd = frac(sin(dot(rip + float2(1, 1), float2(127.1, 311.7))) * 43758.5453);
        float region = lerp(lerp(ra, rb, rf.x), lerp(rc, rd, rf.x), rf.y);   // 0..1
        float hotZone = lerp(0.55, 1.25, region);
        heatC = saturate((veinCore * 1.0 + vein * 0.5) * hotZone + crustEmber * fbm);
        vtC   = saturate((veinCore * 0.75 + vein * 0.3) * lerp(0.78, 1.06, region));
    }
}

// analytic tangent-space normal from the height gradient (central difference)
float nstr = NormalStrength;
float3 OutN = normalize(float3(-(Hs[1] - Hs[0]) * nstr, -(Hs[2] - Hs[0]) * nstr, 1.0));
OutNormal = OutN;

// vein internal temperature -> 4-stop thermal ramp
float3 ember = float3(0.30, 0.02, 0.0);
float3 midc  = float3(0.95, 0.22, 0.0);
float3 hotc  = float3(1.00, 0.65, 0.0);
float3 white = float3(1.00, 0.95, 0.6);
float3 veinCol;
if (vtC < 0.35)      veinCol = lerp(ember, midc, vtC / 0.35);
else if (vtC < 0.70) veinCol = lerp(midc,  hotc, (vtC - 0.35) / 0.35);
else                 veinCol = lerp(hotc,  white, saturate((vtC - 0.70) / 0.30));

return float4(veinCol, heatC);
