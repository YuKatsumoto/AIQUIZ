// Simple molten-LIQUID surface: a near-uniform glowing colour with an animated
// ripple normal (so light/specular shimmers across it like a moving fluid).
// NO crack / crust / voronoi pattern -- the surface reads as a clean solid-colour
// liquid. Inputs: UV(float2), Time(float), RippleScale, RippleStrength (scalars).
// Additional output: OutNormal(float3, tangent space). Return: brightness (float).

float Tw = fmod(Time, 120.0);
float2 p = UV * RippleScale;

// sum of moving directional waves -> a smooth ripple height field
float h =
    sin(p.x * 1.00 + Tw * 1.30) * 0.50 +
    sin(p.y * 1.30 - Tw * 1.10) * 0.40 +
    sin((p.x + p.y) * 0.80 + Tw * 1.70) * 0.30 +
    sin((p.x - p.y) * 1.70 - Tw * 0.90) * 0.20;

// analytic gradient of that height field -> tangent-space ripple normal
float dhx =
    cos(p.x * 1.00 + Tw * 1.30) * 0.50 * 1.00 +
    cos((p.x + p.y) * 0.80 + Tw * 1.70) * 0.30 * 0.80 +
    cos((p.x - p.y) * 1.70 - Tw * 0.90) * 0.20 * 1.70;
float dhy =
    cos(p.y * 1.30 - Tw * 1.10) * 0.40 * 1.30 +
    cos((p.x + p.y) * 0.80 + Tw * 1.70) * 0.30 * 0.80 -
    cos((p.x - p.y) * 1.70 - Tw * 0.90) * 0.20 * 1.70;
OutNormal = normalize(float3(-dhx * RippleStrength, -dhy * RippleStrength, 1.0));

// molten brightness flows with the ripples (crests hot, troughs cooler) so the
// single-colour liquid visibly streams/flows even though it is fully emissive.
float bright = 0.55 + 0.45 * saturate(h * 0.5 + 0.5);
return bright;
